/* 1. Из коллекции Atlas sample_airbnb.listingsAndReviews сгруппировать по цене за ночь недвижимость Барселоны ("address.market": "Barcelona"):
    0-50
    50-100
    100-1000
    Дороже 1000

По каждому ценовому диапазону необходимо получить:
  Количество предложений (документов БД)
  Информацию по каждому объекту:
    Название объекта
    Его цена
  Средняя цена внутри диапазона
*/
db.getCollection('listingsAndReviews').aggregate(
  [
    { $match: { 'address.market': 'Barcelona' } },
    {
      $bucket: {
        groupBy: '$price',
        boundaries: [0, 50, 100, 1000, Infinity],
        default: 'other',
        output: {
          count_docs: { $sum: 1 },
          avg_price: { $avg: '$price' },
          objinfo: {
            $push: {
              obj_name: '$name',
              obj_price: '$price'
            }
          }
        }
      }
    },
    {
      $project: {
        _id: 0,
        count_docs: '$count_docs',
        avg_price: { $round: ['$avg_price', 2] },
        objinfo: 1,
        category: {
          $switch: {
            branches: [
              {
                case: { $eq: ['$_id', 0] },
                then: '0-50'
              },
              {
                case: { $eq: ['$_id', 50] },
                then: '50-100'
              },
              {
                case: { $eq: ['$_id', 100] },
                then: '100-1000'
              },
              {
                case: { $eq: ['$_id', 1000] },
                then: '1000+'
              }
            ],
            default: '_'
          }
        }
      }
    }
  ],
  { maxTimeMS: 60000, allowDiskUse: true }
);




/* 2. Из коллекции sample_airbnb.listingsAndReviews
создать новую коллекцию reviews
и отправить в нее все отзывы из исходной коллекции. 1 документ - 1 отзыв.
*/

/* ВАЖНО: убрать поле _id перед стадией $out,
чтобы MongoDB не перенёс задублированные _id (которые ВСЕГДА дублирует стадия $unwind),
а создал новые собственные */

client['sample_airbnb']['listingsAndReviews'].aggregate([
    {
        '$match': {
            'reviews': {
                '$exists': True, 
                '$ne': []
            }
        }
    }, {
        '$unwind': '$reviews'
    }, {
        '$project': {
            'listing_id': '$reviews.listing_id', 
            'reviewer_id': '$reviews.reviewer_id', 
            'reviewer_name': '$reviews.reviewer_name', 
            'comments': '$reviews.comments', 
            'id': '$_id', 
            '_id': 0
        }
    }, {
        '$out': 'reviwes_from_listingsAndReviews'
    }
])


/* 3. Из коллекции Atlas sample_restaurants.restaurants
определить среднюю оценку каждого ресторана avg_grade (grades.score)
Затем распределить рестораны по диапазонам средней оценки:[0, 20, 40, 60, 80, 100]
Таким образом выяснить группу лучших и худших ресторанов. */

client['sample_restaurants']['restaurants'].aggregate([
    {
        '$project': {
            '_id': 0, 
            'name': 1, 
            'avg_grade': {
                '$avg': '$grades.score'
            }
        }
    }, {
        '$bucket': {
            'groupBy': '$avg_grade', 
            'boundaries': [
                0, 20, 40, 60, 80, 100
            ], 
            'default': 'other', 
            'output': {
                'count': {
                    '$sum': 1
                }, 
                'restaurants': {
                    '$push': {
                        'name': '$name', 
                        'avg_grade': '$avg_grade'
                    }
                }
            }
        }
    }
])

/* 4. Добавить в коллекцию Atlas sample_restaurants.restaurants новые поля:
- avgScore - среднее значение всех оценок для этого ресторана (grades.score),
- grade - последнюю оценку из отзывов (grades.grade) (первый элемент массива)
- totalReviews - общее количество отзывов.
*/

client['sample_restaurants']['restaurants'].aggregate([
    {
        '$set': {
            'avgScore': {
                '$avg': '$grades.score'
            }, 
            'grade': {
                '$first': '$grades.grade'
            }, 
            'totalReviews': {
                '$size': '$grades'
            }
        }
    }, {
        '$out': 'restaurants_agg'
    }
])

/* 5. Из коллекции Atlas sample_supplies.sales:
добавить новое поле - total - итоговая сумма покупки,
которая вычисляется как сумма стоимости всех элементов items,
каждый из которых является произведением количества на стоимость одной единицы.
*/

client['sample_supplies']['sales'].aggregate([
    {
        '$unwind': '$items'
    }, {
        '$set': {
            'amout': {
                '$multiply': [
                    '$items.price', '$items.quantity'
                ]
            }
        }
    }, {
        '$group': {
            '_id': '$_id', 
            'total_amout': {
                '$sum': '$amout'
            }
        }
    }, {
        '$lookup': {
            'from': 'sales', 
            'localField': '_id', 
            'foreignField': '_id', 
            'as': 'result'
        }
    }, {
        '$unwind': '$result'
    }, {
        '$set': {
            'result.total_amout': '$total_amout'
        }
    }, {
        '$replaceRoot': {
            'newRoot': '$result'
        }
    }, {
        '$out': 'sales_with_totale'
    }
])

/* 6. Отсортировать по новому полю.
Узнать, где были совершены самые большие заказы? */


client['sample_supplies']['sales_with_totale'].aggregate([
    {
        '$sort': {
            'total_amout': -1
        }
    }, {
        '$limit': 10
    }, {
        '$project': {
            'storeLocation': 1, 
            'total_amout': 1, 
            '_id': 0
        }
    }
])
