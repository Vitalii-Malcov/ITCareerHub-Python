// 2. Работаем с коллекцией ich_edit.UK_used_cars_ВАШЕ_ИМЯ

// 3. Вывести всех автопроизводителей.
[
    { '$group': { '_id': '$manufacturer' } },
    { '$project': { 'manufacturer': '$_id', '_id': 0 } }
]

// 4. Узнать, по какому из них наибольшее количество документов?
[
    { '$group': { '_id': '$manufacturer', 'cout': { '$sum': 1 } } },
    { '$sort': { 'count': -1 } },
    { '$limit': 1 }
]
/* 5. Сгруппировать по типам трансмиссии, чтобы выяснить, каких машин больше -
 с автоматической и полуавтоматической коробкой передач и с механической? */
[
    { '$group': { '_id': '$transmission', 'count': { '$sum': 1 } } },
    { '$sort': { 'count': -1 } }
]

/* 6. Найти топ 3 производителей с наибольшим средним пробегом в этой коллекции.
Для удобства можно добавить количество документов. */
[
    { '$group': { '_id': '$manufacturer', 'avgMileage': { '$avg': '$mileage' }, 'count': { '$sum': 1 } } },
    { '$sort': {'avgMileage': -1 } },
    { '$limit': 3 }
]

/* 7. Добавить новое поле и перевести mpg в l/100km т.е. из миль на галлон (mpg)
в литры на 100 километров (л/100км)
Формула для конвертации миль на галлон в литры на 100 километров:
l/100km=282.481 /mpg */

/* 8. Добавить новое поле в существующую коллекцию при помощи  $merge.
Примечание:
Чтобы добавить новое поле в существующую коллекцию с помощью $merge,
мы сначала вычисляем новое поле с использованием $addFields,
а затем сливаем результаты обратно в оригинальную коллекцию с помощью $merge.
*/
[
    { '$addFields': { 'l_per_100km': { 'coun': [ { '$and': [ { '$ne': [ '$mpg', None ] } }, { '$gt': [ '$mpg', 0 ] } ] },
    { '$round': [ { '$divide': [ 282.481, '$mpg' ] }, 2 ] }, None ] } } },
    { '$merge': { 'into': 'UK_user_cars_Vitalii_Malcov', 'on': '_id', 'whenMatched': 'merge', 'whenNotMatched': 'discard' } }
]

/* 9. Распределить дизельные авто по цене на категории
 - до 20000,
 - от 20000 до 30000,
 - от 30000 до 40000,
 - от 40000 до 60000,
 - и свыше 60000.
Какая категория сама популярная? */
[
  { $match: { fuelType: "Diesel" } },
  { $bucket: { groupBy: "$price", boundaries: [ 0, 20000, 30000, 40000, 60000, Infinity ],
   default: "Other",
   output: { count: { $sum: 1 } } } },
  { $sort: { count: -1 } }
]

/* 10. Изменить текущие имена (названия) категорий на более удобные. */
[
  { $match: { fuelType: "Diesel" } },
  { $addFields: { priceCategory: { $switch: { branches: [
  { case: { $lt: ["$price", 20000] }, then: "до 20 000" },
  { case: { $lt: ["$price", 30000] }, then: "20 000 - 30 000" },
  { case: { $lt: ["$price", 40000] }, then: "30 000 - 40 000" },
  { case: { $lt: ["$price", 60000] }, then: "40 000 - 60 000" } ],
  default: "свыше 60 000" } } } },
  { $merge: { into: "UK_user_cars_Vitalii_Malcov", on: "_id", whenMatched: "merge", whenNotMatched: "discard" } }
]

[
    { '$match': { 'fuelType': 'Diesel' } },
    { '$project': { '_id': 0, 'manufacturer': 1, 'price': 1, 'priceCategory': 1 } },
    { '$limit': 5 }
]
