/* 2. Посчитать средний возраст клиентов по каждой категории образования. */
[
    {
        '$group': {
            '_id': '$Education_Level',
            'avgAge': {
                '$avg': '$Customer_Age'
            }
        }
    }, {
        '$sort': {
            'avgAge': -1
        }
    }
]
/* 3. Определить средний кредитный лимит по категориям дохода. */
[
    {
        '$group': {
            '_id': '$Income_Category',
            'avgCreditLimit': {
                '$avg': '$Credit_Limit'
            }
        }
    }, {
        '$sort': {
            'avgCreditLimit': -1
        }
    }
]
/* 4. Найти средний возраст клиентов и их средний кредитный лимит для каждого семейного статуса. */
[
    {
        '$group': {
            '_id': '$Marital_Status',
            'avgAge': {
                '$avg': '$Customer_Age'
            },
            'avgCreditLimit': {
                '$avg': '$Credit_Limit'
            }
        }
    }, {
        '$project': {
            'Marital_Status': '$_id',
            'avgAge': {
                '$round': [
                    '$avgAge', 1
                ]
            },
            'avgCreditLimit': {
                '$round': [
                    '$avgCreditLimit'
                ]
            },
            '_id': 0
        }
    }, {
        '$sort': {
            '_id': 1
        }
    }
]
/* 5. Найти самый часто встречающийся уровень образования среди клиентов */

]
