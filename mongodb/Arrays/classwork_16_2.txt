// 1. Сгруппировать по вендорам коллекцию ich.bookings (сколько документов по каждому вендору)
[
  {
    $group: {
      _id: "$vendor",
      bookings_count: {
        $sum: 1
      }
    }
  }
]

// 2. Найти средний показатель километража с разбивкой по вендорам
[
  {
    $group: {
      _id: "$vendor",
      average_km: {
        $avg: "$km"
      }
    }
  },
  {
    $project: {
      _id: 0,
      vendor: "$_id",
      average_km: {
        $round: ["$average_km", 2]
      }
    }
  }
]

// 3. Найти вендора, у которого средний километраж больше 2000 км
[
  {
    $group: {
      _id: "$vendor",
      average_km: {
        $avg: "$km"
      }
    }
  },
  {
    $match: {
      average_km: {
        $gt: 2000
      }
    }
  },
  {
    $project: {
      _id: 0,
      vendor: "$_id",
      average_km: {
        $round: ["$average_km", 2]
      }
    }
  }
]

// 4. Используя коллекцию ich.imdb, сгруппировать количество фильмов (а не сериалов!) по годам.
[
    {
        '$match': {
            'type': 'movie'
        }
    }, {
        '$group': {
            '_id': '$year',
            'count': {
                '$sum': 1
            }
        }
    }
]
/* 5. Сгруппировать количество фильмов по годам между 1980 и 1990.
Отсортировать по годам, по возрастанию. */
[
  {
    $match: {
      type: "movie",
      year: {
        $gte: 1980,
        $lte: 1990
      }
    }
  },
  {
    $group: {
      _id: "$year",
      movies_count: {
        $sum: 1
      }
    }
  },
  {
    $sort: {
      _id: 1
    }
  },
  {
    $project: {
      _id: 0,
      year: "$_id",
      movies_count: 1
    }
  }
]

// 6. Найти 3 самых популярных страны из списка всех фильмов.
[
  {
    $match: {
      type: "movie"
    }
  },
  {
    $unwind: "$countries"
  },
  {
    $group: {
      _id: "$countries",
      movies_count: {
        $sum: 1
      }
    }
  },
  {
    $sort: {
      movies_count: -1
    }
  },
  {
    $limit: 3
  },
  {
    $project: {
      _id: 0,
      country: "$_id",
      movies_count: 1
    }
  }
]
